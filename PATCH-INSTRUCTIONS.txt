Groups reactions:
Use the new `components/ReactionRoster.tsx` in your groups page anywhere emoji reactions are shown.

Pattern:

import { ReactionRoster } from "../../components/ReactionRoster";

const grouped = new Map<string, any[]>();
(message.reactions || []).forEach((r: any) => {
  grouped.set(r.emoji, [...(grouped.get(r.emoji) || []), r]);
});

{EMOJIS.map((emoji) => (
  <ReactionRoster
    key={emoji}
    emoji={emoji}
    reactions={grouped.get(emoji) || []}
    onReact={() => onReact(message.id, emoji)}
    buttonStyle={REACT_SMALL}
  />
))}

This gives you a hover popover with linked usernames instead of a plain title tooltip.
